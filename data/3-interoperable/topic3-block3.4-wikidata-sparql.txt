# What this query does:
# Asks Wikidata: "show me all entities that have both a ROR ID (P6782)
# AND are located in Latvia (P17 = Q211)" — i.e., Latvian research institutions
# registered in ROR.
#
# Pedagogical use: demonstrates SPARQL as the reuse-of-others'-linked-data mechanism;
# bridges to Topic 4 SPARQL hands-on (locked decision #3).

SELECT ?inst ?instLabel ?ror WHERE {
  ?inst wdt:P6782 ?ror .
  ?inst wdt:P17 wd:Q211 .
  SERVICE wikibase:label { bd:serviceParam wikibase:language "lv,en" }
} LIMIT 10
